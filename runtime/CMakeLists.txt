# The metis runtime — the primary build (runtime-binding step 1).
#
#   cmake -S runtime -B runtime/build && cmake --build runtime/build
#   ctest --test-dir runtime/build --output-on-failure
#
# metis_kernel  INTERFACE — the header-only C++17 runtime (include/metis);
#               BUILD.bazel carries the same target for Bazel consumers
# metis_rt      SHARED    — libmetis_rt.so, the C ABI (c/metis_rt.h) a
#               device host or cgo links; lands in the build dir itself
# the eight tools (tools/), and ctest over the committed goldens
# (../corpus/goldens) and the smokes' fixtures (fixtures/, generated by
# metispy — each names its commit and generator hash; never hand-edit).
# No Python and no OCaml are needed here.
cmake_minimum_required(VERSION 3.16)
project(metis_runtime LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_C_STANDARD 99)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "build type" FORCE)
endif()

set(METIS_CORPUS ${CMAKE_CURRENT_SOURCE_DIR}/../corpus/goldens)
set(METIS_FIXTURES ${CMAKE_CURRENT_SOURCE_DIR}/fixtures)

add_library(metis_kernel INTERFACE)
target_include_directories(metis_kernel INTERFACE
  ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_features(metis_kernel INTERFACE cxx_std_17)

add_library(metis_rt SHARED c/metis_rt.cpp)
target_include_directories(metis_rt PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/c)
target_link_libraries(metis_rt PRIVATE metis_kernel)
set_target_properties(metis_rt PROPERTIES
  C_VISIBILITY_PRESET hidden
  CXX_VISIBILITY_PRESET hidden
  VISIBILITY_INLINES_HIDDEN ON
  VERSION 0.1.0
  SOVERSION 0)

foreach(tool metis_run sample_run fw_smoke budget_smoke evaluator_smoke
             ir_eval_smoke live_run project_run)
  add_executable(${tool} tools/${tool}.cpp)
  target_link_libraries(${tool} PRIVATE metis_kernel)
endforeach()

add_executable(metis_rt_conformance tests/conformance.c)
target_link_libraries(metis_rt_conformance PRIVATE metis_rt)
add_executable(metis_rt_abi_smoke tests/abi_smoke.c)
target_link_libraries(metis_rt_abi_smoke PRIVATE metis_rt)

enable_testing()

# the first gate: the goldens through the C ABI (ci/check.sh gate 6's
# IR PARITY 46/46, the ground filter corpus, gate 7's SAMPLE PARITY 43/43)
add_test(NAME conformance
  COMMAND metis_rt_conformance ${METIS_CORPUS}/ir_goldens.txt
                               ${METIS_CORPUS}/goldens.txt
                               ${METIS_CORPUS}/sample_goldens.txt)
# the live handle through the C ABI, in C
add_test(NAME abi_smoke
  COMMAND metis_rt_abi_smoke ${METIS_FIXTURES}/live_cascade.wire)
# the same corpora natively (tools/metis_run, the C++ entry)
add_test(NAME metis_run
  COMMAND metis_run ${METIS_CORPUS}/ir_goldens.txt
                    ${METIS_CORPUS}/goldens.txt)
# the normative sampler on the library (gate 7: SAMPLE PARITY 43/43)
add_test(NAME sample_run
  COMMAND sample_run ${METIS_CORPUS}/sample_goldens.txt)
add_test(NAME fw_smoke COMMAND fw_smoke)
add_test(NAME budget_smoke COMMAND budget_smoke)
add_test(NAME evaluator_smoke
  COMMAND evaluator_smoke ${METIS_FIXTURES}/fw_rover_v1.txt
                          ${METIS_FIXTURES}/fw_rover_v2.txt
                          ${METIS_FIXTURES}/fw_rover_expect.txt
                          ${CMAKE_CURRENT_BINARY_DIR}/fw_rover_serve.txt)
add_test(NAME ir_eval_smoke
  COMMAND ir_eval_smoke ${METIS_FIXTURES}/ir_rover_artifact.txt
                        ${METIS_FIXTURES}/ir_rover_expect.txt
                        ${CMAKE_CURRENT_BINARY_DIR}/ir_rover_serve.txt)

# metis::Live (live.hpp) against metispy's live REPL on public programs:
# K1 + K2 byte for byte (seeded traces with a host's do-lines, forced rows,
# a latched stage revived) and K3 (each sample's snapshot restored into a
# reloaded Live runs on identically) — live_run; K4 (the likelihood from a
# state, the watch's surprise test) against surprise_record — project_run
foreach(case cascade love_triangle weights)
  add_test(NAME live_${case}
    COMMAND live_run ${METIS_FIXTURES}/live_${case}.wire
                     ${METIS_FIXTURES}/live_${case}.txt)
  add_test(NAME project_${case}
    COMMAND project_run ${METIS_FIXTURES}/live_${case}.wire
                        ${METIS_FIXTURES}/proj_${case}.txt)
endforeach()

# every tool exits nonzero unless ok == total > 0 (or 0 failures); a
# printed first failure fails the test even so
set_tests_properties(conformance abi_smoke metis_run sample_run fw_smoke budget_smoke
                     evaluator_smoke ir_eval_smoke
                     live_cascade live_love_triangle live_weights
                     project_cascade project_love_triangle project_weights
                     PROPERTIES
  FAIL_REGULAR_EXPRESSION "FIRST FAIL|status [0-9]+:")
